### PP1 — assemble one valid positive chunk
Combine already-valid child evidence without changing the fixed Star rule.

```dafny
module R11_ParallelChunk {
 import opened RG_A1_Re
 import opened RG_A4_Norm
 lemma ChunkR11(r:Re,s:seq<char>,k:nat,a:Derivation,b:Derivation)
  requires 0<k<=|s| && CheckDeriv(r,s[..k],a) && CheckDeriv(Star(r),s[k..],b)
  ensures CheckDeriv(Star(r),s,DStarCons(k,a,b))
 {}
}
```

```rust
use vstd::prelude::*;
use rg_re_a1::*;
use rg_re_a4::*;
verus! {
proof fn chunk_r11(r:Re,s:Seq<char>,k:usize,a:Derivation,b:Derivation)
 requires 0<k<=s.len(),valid_word_r10(a,r,s.take(k as int)),
 valid_word_r10(b,Re::Star(Box::new(r)),s.skip(k as int)),
 ensures valid_word_r10(Derivation::DStarCons(k,Box::new(a),Box::new(b)),Re::Star(Box::new(r)),s),
 {}
}
```
