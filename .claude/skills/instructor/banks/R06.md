### PP2 — compose a sequence translation
**Isolates:** a general recursive hypothesis and an explicit calculation.
Keep the function fixed; use a calc chain and justify the recursive instance.
```dafny
module R06_ParallelBump {
  ghost function BumpR06(s: seq<int>,k: int): seq<int>
    decreases |s|
  { if s == [] then [] else [s[0]+k]+BumpR06(s[1..],k) }
  lemma {:induction false} BumpComposeR06(s: seq<int>,a: int,b: int)
    ensures BumpR06(BumpR06(s,a),b) == BumpR06(s,a+b)
    decreases |s|
}
```

```rust
use vstd::prelude::*;
use vstd::calc;
verus! {
pub open spec fn bump_r06(s: Seq<int>,k: int) -> Seq<int>
    decreases s.len(),
{ if s.len() == 0 { Seq::empty() } else { seq![s[0]+k]+bump_r06(s.subrange(1,s.len() as int),k) } }
proof fn lemma_bump_compose_r06(s: Seq<int>,a: int,b: int)
    ensures bump_r06(bump_r06(s,a),b) == bump_r06(s,a+b),
    decreases s.len(),

}
```
