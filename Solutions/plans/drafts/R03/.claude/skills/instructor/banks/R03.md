### PP1 — sum two opaque tokens
**Isolates:** `module-visibility`, `requires-ensures`, `spec-vs-exec`.

Implement the client sum through the exported operations. Preserve the hidden
representation. Rust must return the widened sum without strengthening inputs.

```dafny
module R03_ParallelTokens {
  module API {
    export provides Token, TokenModel, Issue, Read
    datatype Token = Token(value: nat)
    function TokenModel(t: Token): nat { t.value }
    function Issue(n: nat): (t: Token) ensures TokenModel(t) == n { Token(n) }
    method Read(t: Token) returns (n: nat) ensures n == TokenModel(t) { n := t.value; }
  }
  module Client {
    import opened API
    method SumTokens(a: Token, b: Token) returns (n: nat)
      ensures n == TokenModel(a) + TokenModel(b)
    { assume false; }
  }
}
```

### PP2 — exchange a parcel's observations
**Isolates:** `module-visibility`, `ghost-and-modes`, `frozen-statements`.

Implement the client exchange through the given accessors and constructor;
retain the private representation and the exact observational postcondition.

```dafny
module R03_ParallelParcel {
  module API {
    export provides Parcel, Model, Make, First, Second
    datatype Parcel = Parcel(left: int, right: int)
    function Model(p: Parcel): (int,int) { (p.left,p.right) }
    function Make(a: int,b: int): (p: Parcel) ensures Model(p) == (a,b) { Parcel(a,b) }
    method First(p: Parcel) returns (a: int) ensures a == Model(p).0 { a := p.left; }
    method Second(p: Parcel) returns (b: int) ensures b == Model(p).1 { b := p.right; }
  }
  module Client {
    import opened API
    method Exchange(p: Parcel) returns (r: Parcel)
      ensures Model(r) == (Model(p).1,Model(p).0)
    { assume false; }
  }
}
```

```rust

verus! {
pub mod api {
  use vstd::prelude::*;
  pub struct Parcel { left: u32, right: u32, }
  impl View for Parcel {

  }

    pub fn second(&self) -> (b: u32) ensures b == self@.1, { self.right }
  }
}

  #[verifier::external_body]

}
}
```
